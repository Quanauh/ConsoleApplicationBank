    //Tài khoản khách hàng
    class BankAccount
    {
        private string tkKhachHang;
        private string matkhau;
        private decimal Sodu;
        public BankAccount(string a,string mk)
        {
            this.tkKhachHang=a;
            this.Sodu=0;
            this.matkhau=mk;
        }
        public string getTk()
        {
            return tkKhachHang;
        }
        public decimal getSd()
        {
            return Sodu;
        }
        public void Naptien(decimal a)
        {
            Sodu+=a;
        }
        public void rt(decimal a)
        {
            Sodu-=a;
        }
        public bool checkMk(string mk)
        {
            return matkhau==mk;
        }
    }
    //Quản lý ngân hàng
    class BankManager
    {
        private Dictionary<string,BankAccount> ds= new Dictionary<string,BankAccount>();
        public bool Timstk(string stk)
        {
            return ds.ContainsKey(stk);
        }
        //Tạo tài khoản
        public void TaoTk(string stk,string mk)
        {
            if (Timstk(stk) == false)
            {
                //Số tài khoản sẽ định danh cho 1 tài khoản
                ds[stk]=new BankAccount(stk,mk);
                Console.WriteLine("Tao tai khoan thanh cong");
            }
            else
            {
                Console.WriteLine("So tai khoan da ton tai");
            }        
        }
        public bool checkDangNhap(string stk,string mk)
        {
            return Timstk(stk)&&ds[stk].checkMk(mk);
        }
        //Xử lý nạp tiền
        public void Naptien(string stk,decimal a)
        {
            if (a <= 0)
            {
                Console.WriteLine("So tien khong hop le");
                return;
            }
            else{
            Console.WriteLine("Nap tien thanh cong");
            ds[stk].Naptien(a);
            }
        }
        //Xử lý rút tiền
        public void RutTien(string stk,decimal a)
        {
            if (a <= 0)
            {
                Console.WriteLine("So tien khong hop le");
                return;
            }
            else if (a > ds[stk].getSd())
            {
                Console.WriteLine("So du khong du");
                return;
            }
            else
            {
                Console.WriteLine("Rut tien thanh cong");
                ds[stk].rt(a);
            }
        }
        //Xử lý chuyển tiền
        public void ChuyenTien(string stk1, string stk2,decimal a)
        {
            if (a <= 0)
            {
                Console.WriteLine("So tien khong hop le");
                return;
            }
            if (a > ds[stk1].getSd())
            {
                Console.WriteLine("So du cua ban khong du");
                return;
            }
            else{
            ds[stk1].rt(a);
            ds[stk2].Naptien(a);
            Console.WriteLine("Chuyen tien thanh cong");
            Console.WriteLine($"So du cua ban la: {ds[stk1].getSd()}");
            }
        }
        //Xử lý xem số dư
        public decimal XemSodu(string stk)
        {
            return ds[stk].getSd();
        }
    }
